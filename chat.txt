<script>
const USERS="codechat_users";
let activeCode=null;
let pendingCode=null;

function users(){
 try{return JSON.parse(localStorage.getItem(USERS)||"{}")}
 catch{return {}}
}
function saveUsers(data){
 localStorage.setItem(USERS,JSON.stringify(data));
}
function generateCode(){
 const chars="ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
 const bytes=crypto.getRandomValues(new Uint8Array(12));
 let s="";
 bytes.forEach((b,i)=>{
  if(i===4||i===8)s+="-";
  s+=chars[b%chars.length];
 });
 return s;
}
function register(){
 const name=document.getElementById("username").value.trim();
 if(!name){status("Enter a username.");return}
 const all=users();
 let code;
 do{code=generateCode()}while(all[code]);
 all[code]={name:name};
 saveUsers(all);
 pendingCode=code;
 document.getElementById("createdCode").textContent=code;
 document.getElementById("newAccount").classList.remove("hidden");
 document.getElementById("register").classList.add("hidden");
 status("Account created. Save your code!");
}
function status(message){
 document.getElementById("status").textContent=message;
}
async function copyCode(){
 try{
  await navigator.clipboard.writeText(pendingCode);
  status("Code copied!");
 }catch{
  status("Please select and copy your code manually.");
 }
}
function continueLogin(){
 if(pendingCode)openChat(pendingCode);
}
function login(){
 const code=document.getElementById("loginCode").value.trim().toUpperCase();
 if(!users()[code]){
  status("Account not found in this browser.");
  return;
 }
 openChat(code);
}
function openChat(code){
 const account=users()[code];
 if(!account)return;
 activeCode=code;
 localStorage.setItem("codechat_session",code);
 document.getElementById("auth").style.display="none";
 document.getElementById("chat").style.display="block";
 document.getElementById("welcome").textContent="Hello, "+account.name+"!";
 renderMessages();
}
function logout(){
 activeCode=null;
 localStorage.removeItem("codechat_session");
 document.getElementById("auth").style.display="block";
 document.getElementById("chat").style.display="none";
 document.getElementById("register").classList.remove("hidden");
 document.getElementById("newAccount").classList.add("hidden");
}
function renderMessages(){
 const box=document.getElementById("messages");
 box.replaceChildren();
 const key="codechat_messages_"+activeCode;
 let list=[];
 try{list=JSON.parse(localStorage.getItem(key)||"[]")}catch{}
 if(!list.length){
  box.textContent="No messages yet. Send your first message!";
  return;
 }
 list.forEach(item=>{
  const div=document.createElement("div");
  div.className="message";
  const heading=document.createElement("strong");
  heading.textContent=item.name;
  const text=document.createElement("div");
  text.textContent=item.text;
  const time=document.createElement("small");
  time.textContent=new Date(item.time).toLocaleString();
  div.append(heading,document.createElement("br"),text,time);
  box.appendChild(div);
 });
 box.scrollTop=box.scrollHeight;
}
document.getElementById("messageForm").addEventListener("submit",e=>{
 e.preventDefault();
 if(!activeCode)return;
 const field=document.getElementById("message");
 const text=field.value.trim();
 if(!text)return;
 const key="codechat_messages_"+activeCode;
 let list=[];
 try{list=JSON.parse(localStorage.getItem(key)||"[]")}catch{}
 list.push({name:users()[activeCode].name,text,time:Date.now()});
 localStorage.setItem(key,JSON.stringify(list));
 field.value="";
 renderMessages();
});
try{
 const code=localStorage.getItem("codechat_session");
 if(code&&users()[code])openChat(code);
}catch{}
</script>
